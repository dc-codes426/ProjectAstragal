\noindent
\begin{minipage}[t]{0.52\textwidth}
\textbf{\large BIP-39 Entropy Generation}

\vspace{3pt}
\small
Roll a d6. If \textbf{5 or 6}, discard and re-roll.
When the result is \textbf{1--4}, write it in the next
cell of the top row. Then write its two bits in the pair
of cells directly below it. Fill all 128 top cells
left to right, strip by strip. The bottom rows are
your 256-bit entropy --- read them off in order.
\end{minipage}\hfill
\begin{minipage}[t]{0.18\textwidth}
\centering
\small\bfseries Lookup Key\\[3pt]
\begin{tabular}{|c|c|c|}
\hline
\textsf{Die} & \textsf{bit} & \textsf{bit} \\
\hline
1 & 0 & 0 \\
2 & 0 & 1 \\
3 & 1 & 0 \\
4 & 1 & 1 \\
\hline
\multicolumn{3}{|c|}{\small 5,\,6\,=\,re-roll} \\
\hline
\end{tabular}
\end{minipage}\hfill
\begin{minipage}[t]{0.24\textwidth}
\footnotesize
\textbf{Properties:} 128 kept rolls
$\times$ 2 bits = 256 bits.
Zero arithmetic, zero bias.
$\sim$192 rolls expected.\\[3pt]
\textbf{After:} Enter 256 bits into
air-gapped BIP-39 tool for SHA-256
checksum \& word lookup.
\textbf{Destroy this sheet.}
\end{minipage}
